Display Speck 2f DVS Events

An example that uses a Speck2fDevKit/Speck2fModuleDevkit to display output spike events of speck2f model in visualizer.

A step-by-step example on how to visualize Speck 2f DVS events can be found in Hands On Speck 2f Development Kit.

 1import samna
 2import sys
 3import os
 4from threading import Thread
 5
 6def open_speck2f_dev_kit():
 7    devices = [
 8        device
 9        for device in samna.device.get_unopened_devices()
10        if device.device_type_name.startswith("Speck2f")
11    ]
12    assert devices, "Speck2f board not found"
13
14    return samna.device.open_device(devices[0])
15
16
17def build_samna_event_route(dk, graph, endpoint):
18    # build a graph in samna to show dvs
19    _, _, streamer = graph.sequential(
20        [dk.get_model_source_node(), "Speck2fDvsToVizConverter", "VizEventStreamer"]
21    )
22
23    config_source, _ = graph.sequential([samna.BasicSourceNode_ui_event(), streamer])
24
25    streamer.set_streamer_endpoint(endpoint)
26    if streamer.wait_for_receiver_count() == 0:
27        raise Exception(f'connecting to visualizer on {endpoint} fails')
28
29    visualizer_config = samna.ui.VisualizerConfiguration(
30        plots=[samna.ui.ActivityPlotConfiguration(128, 128, "DVS Layer")]
31    )
32
33    config_source.write([visualizer_config])
34
35
36def open_visualizer(width_proportion, height_proportion, receiver_endpoint):
37    # start visualizer in a isolated process which is required on mac, instead of a sub process.
38    gui_cmd = f"import samna, samnagui; samnagui.run_visualizer('{receiver_endpoint}', {width_proportion}, {height_proportion})"
39    os_cmd = f'{sys.executable} -c "{gui_cmd}"'
40    print("Visualizer start command: ", os_cmd)
41    gui_thread = Thread(target=os.system, args=(os_cmd,))
42    gui_thread.start()
43
44    return gui_thread
45
46
47streamer_endpoint = "tcp://0.0.0.0:40001"
48width_proportion = 0.75
49height_proportion = 0.75
50
51gui_process = open_visualizer(width_proportion, height_proportion, streamer_endpoint)
52
53dk = open_speck2f_dev_kit()
54
55# route events
56graph = samna.graph.EventFilterGraph()
57build_samna_event_route(dk, graph, streamer_endpoint)
58
59graph.start()
60
61# modify configuration
62config = samna.speck2f.configuration.SpeckConfiguration()
63# enable dvs event monitoring
64config.dvs_layer.monitor_enable = True
65dk.get_model().apply_configuration(config)
66
67# wait until visualizer window destroys.
68gui_process.join()
69
70graph.stop()
